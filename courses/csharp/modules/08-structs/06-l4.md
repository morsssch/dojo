# operator + и компания

Оператор — это статический метод с особым именем. Бинарный получает два параметра, унарный — один. Хотя бы один параметр должен быть твоего типа.

```cs run
readonly struct Vec
{
    public double X { get; }
    public double Y { get; }
    public Vec(double x, double y) { X = x; Y = y; }

    public static Vec operator +(Vec a, Vec b) => new Vec(a.X + b.X, a.Y + b.Y);   // бинарный
    public static Vec operator -(Vec a, Vec b) => new Vec(a.X - b.X, a.Y - b.Y);
    public static Vec operator -(Vec a) => new Vec(-a.X, -a.Y);                    // унарный минус
    public static Vec operator +(Vec a) => a;                                       // унарный плюс
    public static Vec operator *(Vec a, double k) => new Vec(a.X * k, a.Y * k);
    public static Vec operator *(double k, Vec a) => a * k;                         // чтобы работало и 2 * v

    public override string ToString() => $"({X}; {Y})";
}

class Program
{
    static void Main()
    {
        var a = new Vec(1, 2);
        var b = new Vec(3, -1);
        Console.WriteLine(a + b);
        Console.WriteLine(a - b);
        Console.WriteLine(-a);
        Console.WriteLine(2 * a + b * 0.5);
        var c = a;
        c += b;                  // += работает сам, если есть +
        Console.WriteLine(c);
    }
}
```

| Правило |  |
| --- | --- |
| Всегда `public static` | иначе CS0558 |
| Парами | `==`/`!=`, ``, `=` |
| `+=`, `-=` | отдельно не пишутся — выводятся из `+` и `-` |
| Порядок операндов | `v * 2` и `2 * v` — разные операторы, нужны оба |


# Под капотом: оператор — это метод

Для компилятора `a + b` с твоими типами — просто вызов статического метода со специальным именем. В IL (помнишь [модуль 1](lesson:dotnet/l2)) никакого «плюса для векторов» нет, есть `call Vec.op_Addition(a, b)`. Это можно увидеть через рефлексию:

```cs run
readonly struct Vec
{
    public int X { get; }
    public Vec(int x) => X = x;
    public static Vec operator +(Vec a, Vec b) => new Vec(a.X + b.X);
    public static Vec operator -(Vec a) => new Vec(-a.X);
}

class Program
{
    static void Main()
    {
        foreach (var m in typeof(Vec).GetMethods())
            if (m.IsSpecialName && m.Name.StartsWith("op_"))
                Console.WriteLine(m.Name);

        var a = new Vec(2);
        var b = new Vec(3);
        Vec c = a + b;          // компилятор пишет: Vec.op_Addition(a, b)
        Console.WriteLine(c.X);
    }
}
```

| Оператор | Имя метода | Оператор | Имя метода |
| --- | --- | --- | --- |
| `a + b` | op_Addition | `-a` | op_UnaryNegation |
| `a - b` | op_Subtraction | `+a` | op_UnaryPlus |
| `a * b` | op_Multiply | `a == b` | op_Equality |
| `a / b` | op_Division | `a != b` | op_Inequality |

Из этого следуют все правила:

- **Почему static.** У оператора нет «главного» операнда: в `a + b` оба равноправны, поэтому нет и `this`, оба приходят параметрами.

- **Почему `v * 2` и `2 * v` — разные операторы.** Это два метода с разными типами параметров: `(Vec, int)` и `(int, Vec)`. Обычная перегрузка методов из [модуля 5](lesson:methods/l4): компилятор выбирает метод по типам аргументов.

- **Почему унарный и бинарный минус уживаются.** Тоже перегрузка: один параметр против двух.

- **Почему `+=` писать не нужно.** `a += b` компилятор разворачивает в `a = a + b` и вызывает твой op_Addition.

# Преобразования: implicit и explicit

Можно научить компилятор превращать твой тип в другой и обратно.

- `implicit` — неявное, когда преобразование безопасно и ничего не теряет: целое число 3 → дробь 3/1.

- `explicit` — только через явное приведение `(double)r`, когда возможна потеря: дробь 1/3 → 0,333….

```cs run
readonly struct Frac
{
    public int N { get; } public int D { get; }
    public Frac(int n, int d) { N = n; D = d; }
    public static implicit operator Frac(int value) => new Frac(value, 1);
    public static explicit operator double(Frac f) => (double)f.N / f.D;
    public override string ToString() => $"{N}/{D}";
}

class Program
{
    static void Main()
    {
        Frac f = 3;                         // implicit
        Console.WriteLine(f);
        double d = (double)new Frac(1, 3);  // explicit
        Console.WriteLine(d);
    }
}
```

В твоём варианте Лабы 1 преобразования не обязательны, но implicit из `int` — хороший бонус: `r + 1` заработает сам.


# Задача: комплексные числа


````task
id: step-5
lang: cs
title: 'Задача: комплексные числа'
prompt: |-
  Опиши `readonly struct Complex` со свойствами `Re`, `Im` (`double`), операторами бинарными `+ - * /`, унарными `+ -`, `==`/`!=` (плюс Equals и GetHashCode), и `ToString()`: `a+bi` или `a-bi` (например `3-2i`). Деление на ноль (0+0i) — `DivideByZeroException`.

  Формулы: (a+bi)(c+di) = (ac − bd) + (ad + bc)i; (a+bi)/(c+di) = ((ac + bd) + (bc − ad)i) / (c² + d²).
starter: |-
  readonly struct Complex
  {
  }
solution: |-
  readonly struct Complex : IEquatable<Complex>
  {
      public double Re { get; }
      public double Im { get; }
      public Complex(double re, double im) { Re = re; Im = im; }

      public static Complex operator +(Complex a, Complex b) => new(a.Re + b.Re, a.Im + b.Im);
      public static Complex operator -(Complex a, Complex b) => new(a.Re - b.Re, a.Im - b.Im);
      public static Complex operator *(Complex a, Complex b) => new(a.Re * b.Re - a.Im * b.Im, a.Re * b.Im + a.Im * b.Re);
      public static Complex operator /(Complex a, Complex b)
      {
          double den = b.Re * b.Re + b.Im * b.Im;
          if (den == 0) throw new DivideByZeroException();
          return new((a.Re * b.Re + a.Im * b.Im) / den, (a.Im * b.Re - a.Re * b.Im) / den);
      }
      public static Complex operator -(Complex a) => new(-a.Re, -a.Im);
      public static Complex operator +(Complex a) => a;

      public bool Equals(Complex o) => Re == o.Re && Im == o.Im;
      public override bool Equals(object? obj) => obj is Complex c && Equals(c);
      public override int GetHashCode() => HashCode.Combine(Re, Im);
      public static bool operator ==(Complex a, Complex b) => a.Equals(b);
      public static bool operator !=(Complex a, Complex b) => !a.Equals(b);

      public override string ToString() => Im >= 0 ? $"{Re}+{Im}i" : $"{Re}{Im}i";
  }
tests:
  - name: арифметика
    stdout: |-
      4+1i
      -2+3i
      5+5i
      1+2i
      -1-2i
      1+2i
    harness: |-
      var a = new Complex(1, 2);
      var b = new Complex(3, -1);
      Console.WriteLine(a + b);
      Console.WriteLine(a - b);
      Console.WriteLine(a * b);
      Console.WriteLine((a * b) / b);
      Console.WriteLine(-a);
      Console.WriteLine(+a);
  - name: равенство и ноль
    stdout: |-
      True
      True
      ok
    harness: |-
      Console.WriteLine(new Complex(1, 1) == new Complex(1, 1));
      Console.WriteLine(new Complex(1, 1) != new Complex(1, 2));
      try { var x = new Complex(1, 1) / new Complex(0, 0); Console.WriteLine("нет"); } catch (DivideByZeroException) { Console.WriteLine("ok"); }
hints:
  - 'ToString: Im >= 0 ? $"{Re}+{Im}i" : $"{Re}{Im}i" — у отрицательного Im минус выведется сам.'
````
