solution: |-
  int n = int.Parse(Console.ReadLine());
  var pts = new Point[n];
  for (int i = 0; i < n; i++)
  {
      var p = Console.ReadLine().Split(' ');
      pts[i] = new Point(int.Parse(p[0]), int.Parse(p[1]));
  }

  int best = n > 0 ? 1 : 0;
  for (int i = 0; i < n; i++)
  {
      var count = new Dictionary<Dir, int>();
      for (int j = i + 1; j < n; j++)
      {
          Dir d = pts[j] - pts[i];
          count[d] = count.TryGetValue(d, out int c) ? c + 1 : 1;
          best = Math.Max(best, count[d] + 1);
      }
  }
  Console.WriteLine(best);

  readonly struct Point
  {
      public int X { get; }
      public int Y { get; }
      public Point(int x, int y) { X = x; Y = y; }
      public static Dir operator -(Point a, Point b) => new Dir(a.X - b.X, a.Y - b.Y);
  }

  readonly struct Dir : IEquatable<Dir>
  {
      public int Dx { get; }
      public int Dy { get; }

      public Dir(int dx, int dy)
      {
          if (dx == 0 && dy == 0) throw new ArgumentException("Точки совпадают");
          int g = Gcd(dx, dy);
          dx /= g;
          dy /= g;
          if (dx < 0 || (dx == 0 && dy < 0)) { dx = -dx; dy = -dy; }
          Dx = dx;
          Dy = dy;
      }

      static int Gcd(int a, int b)
      {
          a = Math.Abs(a);
          b = Math.Abs(b);
          while (b != 0) (a, b) = (b, a % b);
          return a;
      }

      public bool Equals(Dir o) => Dx == o.Dx && Dy == o.Dy;
      public override bool Equals(object? obj) => obj is Dir d && Equals(d);
      public override int GetHashCode() => HashCode.Combine(Dx, Dy);
      public static bool operator ==(Dir a, Dir b) => a.Equals(b);
      public static bool operator !=(Dir a, Dir b) => !a.Equals(b);
  }
