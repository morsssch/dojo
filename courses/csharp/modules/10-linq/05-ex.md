starter: |-

solution: |-
  class TextPipeline
  {
      private readonly List<TextTransformer> transformers = new();
      private readonly List<TextValidator> validators = new();

      public event Action<string>? ValidationFailed;
      public event Action<string, string>? TransformationApplied;

      public void AddTransformer(TextTransformer t) => transformers.Add(t ?? throw new ArgumentNullException(nameof(t)));
      public void AddValidator(TextValidator v) => validators.Add(v ?? throw new ArgumentNullException(nameof(v)));

      public string? Process(string text)
      {
          if (text == null) throw new ArgumentNullException(nameof(text));
          foreach (var v in validators)
              if (!v(text)) { ValidationFailed?.Invoke(text); return null; }
          foreach (var t in transformers)
          {
              string next = t(text);
              TransformationApplied?.Invoke(text, next);
              text = next;
          }
          return text;
      }

      public void ClearPipeline() { transformers.Clear(); validators.Clear(); }
  }
tests:
  - name: основная проверка
    harness: |-

      p.AddTransformer(s => s.ToUpper());
      Console.WriteLine(p.Process("abc"));

    stdout: |-

  static class TextExtensions
  {
      public static IEnumerable<string> GetWords(this string text)
      {
          if (text == null) throw new ArgumentNullException(nameof(text));
          var sb = new StringBuilder();
          foreach (char c in text)
          {
              if (char.IsLetterOrDigit(c)) sb.Append(c);
              else if (sb.Length > 0) { yield return sb.ToString(); sb.Clear(); }
          }
          if (sb.Length > 0) yield return sb.ToString();
      }

      public static string RemovePunctuation(this string text)
      {
          if (text == null) throw new ArgumentNullException(nameof(text));
          return new string(text.Where(c => !char.IsPunctuation(c)).ToArray());
      }

      public static string Capitalize(this string text)
      {
          if (text == null) throw new ArgumentNullException(nameof(text));
          var chars = text.ToCharArray();
          for (int i = 0; i < chars.Length; i++)
              if (char.IsLetter(chars[i]) && (i == 0 || !char.IsLetterOrDigit(chars[i - 1])))
                  chars[i] = char.ToUpper(chars[i]);
          return new string(chars);
      }

      public static string ReverseWords(this string text)
      {
          if (text == null) throw new ArgumentNullException(nameof(text));
          return string.Join(" ", text.Split(' ', StringSplitOptions.RemoveEmptyEntries).Reverse());
      }

      public static int WordCount(this string text) => text.GetWords().Count();
  }

  static class TextProcessors
  {
      private static readonly string[] Digits = { "ноль", "один", "два", "три", "четыре", "пять", "шесть", "семь", "восемь", "девять" };
      private const string Allowed = ".,!?;:-()'\"";

      public static TextTransformer ToUpperCase() => s => s.ToUpper();
      public static TextTransformer ToLowerCase() => s => s.ToLower();
      public static TextTransformer RemoveExtraSpaces() => s => string.Join(" ", s.Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries));
      public static TextTransformer ReplaceDigitsWithWords() => delegate (string s)
      {
          var sb = new StringBuilder();
          foreach (char c in s) sb.Append(char.IsDigit(c) ? Digits[c - '0'] : c.ToString());
          return sb.ToString();
      };

      public static TextValidator MinLength(int length) => s => s.Length >= length;
      public static TextValidator MaxLength(int length) => s => s.Length <= length;
      public static TextValidator ContainsWord(string word)
      {
          if (word == null) throw new ArgumentNullException(nameof(word));
          return s => s.GetWords().Any(w => string.Equals(w, word, StringComparison.OrdinalIgnoreCase));
      }
      public static TextValidator NoSpecialCharacters() => s => s.All(c => char.IsLetterOrDigit(c) || char.IsWhiteSpace(c) || Allowed.Contains(c));
  }

solution: |-
  class TextAnalyzer
  {
      public Dictionary<string, int> GetWordFrequency(string text) =>
          text.GetWords().Select(w => w.ToLower()).GroupBy(w => w).ToDictionary(g => g.Key, g => g.Count());

      public IEnumerable<string> GetLongestWords(string text, int count) =>
          text.GetWords().Select(w => w.ToLower()).Distinct().OrderByDescending(w => w.Length).Take(count);

      public char GetMostFrequentChar(string text)
      {
          var letters = text.Where(char.IsLetter).Select(char.ToLower).ToList();
          if (letters.Count == 0) throw new InvalidOperationException("Нет букв");
          return letters.GroupBy(c => c).OrderByDescending(g => g.Count()).First().Key;
      }
  }

  > **Конвейер обработки текстовых данных.** Разработать систему конвейерной обработки текстовых данных с использованием цепочки делегатов и методов расширения.
  > 
  > - Делегаты: `TextTransformer` — получает и возвращает строку; `TextValidator` — получает строку, возвращает логический признак корректности.
  > 
  > - Класс `TextPipeline`. Поля: `transformers` — список функций преобразования, `validators` — список валидаторов. События: `ValidationFailed` — при неудачной валидации, параметр — строка, не прошедшая проверку; `TransformationApplied` — после каждого преобразования, получает исходный и результирующий текст. Методы: `AddTransformer`, `AddValidator`, `Process` — последовательно применяет все валидаторы и преобразователи и возвращает результат, при ошибке валидации возвращает `null`; `ClearPipeline` — очищает оба списка.
  > 
  > - Статический класс `TextExtensions`, расширения для `string`: `RemovePunctuation`, `Capitalize` (первая буква каждого слова заглавная), `ReverseWords` (обратный порядок слов), `WordCount`, `GetWords` (последовательность слов).
  > 
  > - Статический класс `TextProcessors`. Преобразователи (возвращают TextTransformer): `ToUpperCase`, `ToLowerCase`, `RemoveExtraSpaces`, `ReplaceDigitsWithWords` (0 → «ноль», 1 → «один» и т. д.). Валидаторы (возвращают TextValidator): `MinLength(int length)`, `MaxLength(int length)`, `ContainsWord(string word)`, `NoSpecialCharacters()`.
  > 
  > - Класс `TextAnalyzer` на LINQ: `GetWordFrequency` — словарь частот слов; `GetLongestWords` — указанное количество самых длинных слов (количество и текст — параметры); `GetMostFrequentChar` — наиболее частый символ.
  > 
  > - Демо: конвейер с несколькими преобразователями, разные валидаторы, подписка на события, обработка разных строк, методы расширения, **анонимные методы и лямбды** для кастомных преобразователей и валидаторов, анализ текста на LINQ.

  Договорённости для автопроверки (это и твои проектные решения — озвучь их на защите)

  - **Слово** — непрерывная последовательность букв и цифр (`char.IsLetterOrDigit`). `GetWords`, `WordCount`, `ContainsWord`, анализатор работают с этим определением.

  - Методы `TextProcessors` — методы **без параметров или с параметром**, возвращающие делегат: `TextProcessors.ToUpperCase()`.

  - События: `event Action? ValidationFailed`, `event Action? TransformationApplied`.

  - `Process(null)` → `ArgumentNullException`. Валидаторы проверяются **до** преобразований, на исходном тексте; первая неудача → событие и `null`.

  - `RemovePunctuation` удаляет символы, для которых `char.IsPunctuation` истинно. `Capitalize` меняет только первую букву каждого слова. `ReverseWords` делит по пробелам и склеивает одним пробелом (знаки остаются при словах).

  - `RemoveExtraSpaces`: подряд идущие пробельные символы → один пробел, плюс Trim. `ReplaceDigitsWithWords`: каждая цифра заменяется словом прямо на месте: `a1b` → `aодинb`.

  - `ContainsWord` — без учёта регистра, целым словом. `NoSpecialCharacters`: разрешены буквы, цифры, пробельные символы и знаки `.,!?;:-()'"`, всё остальное — спецсимвол.

  - Анализатор: слова в нижнем регистре. `GetLongestWords(text, count)` — различные слова по убыванию длины, при равной — в порядке появления. `GetMostFrequentChar` — самая частая **буква** в нижнем регистре, при равенстве — встретившаяся раньше; нет букв → `InvalidOperationException`.

  - Демо — в `class Program { static void Main() }`.
